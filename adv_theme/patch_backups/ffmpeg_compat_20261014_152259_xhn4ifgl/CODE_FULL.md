## rtl/dash_booth.sv

```systemverilog
`timescale 1ns/1ps
// V18A correction-vector radix-4 method, generalized balanced tree for BW=6..12.
module dash_booth #(parameter AW=12, BW=8)(
 input wire signed [AW-1:0] a, input wire signed [BW-1:0] b,
 output wire signed [AW+BW-1:0] p);
 localparam NG=(BW+1)/2, CW=AW+BW+2, NP=2**$clog2(NG);
 reg [2*NG:0] y;
 reg signed [CW-1:0] ae,base;
 reg signed [CW-1:0] pp[0:NG-1];
 reg [CW-1:0] corr;
 reg [2:0] code;
 integer i;
 always @* begin
  y=0;y[0]=0;corr=0;ae={{(CW-AW){a[AW-1]}},a};base=0;code=0;
  for(i=0;i<BW;i=i+1)y[i+1]=b[i];
  for(i=BW;i<2*NG;i=i+1)y[i+1]=b[BW-1];
  for(i=0;i<NG;i=i+1)begin
   code=y[2*i+:3];
   case(code)
    3'b001,3'b010,3'b101,3'b110:base=ae;
    3'b011,3'b100:base=ae<<<1;
    default:base=0;
   endcase
   if(code==3'b100 || code==3'b101 || code==3'b110)begin pp[i]=(~base)<<<(2*i);corr[2*i]=1;end
   else pp[i]=base<<<(2*i);
  end
 end
 wire signed [CW-1:0] tree[0:2*NP-2];
 genvar k;
 generate for(k=0;k<NP;k=k+1)begin
  if(k<NG)assign tree[NP-1+k]=pp[k];else assign tree[NP-1+k]=0;
 end
 for(k=0;k<NP-1;k=k+1)begin assign tree[k]=tree[2*k+1]+tree[2*k+2];end endgenerate
 wire signed [CW-1:0] total=tree[0]+$signed(corr);
 assign p=total[AW+BW-1:0];
endmodule
```

```python
"""Small shared I/O helpers; no implicit working-directory dependencies."""
import hashlib
import json
import os
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]

def sha256(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda: f.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

def write_json(path, value):
    p = Path(path); p.parent.mkdir(parents=True, exist_ok=True)
    t = p.with_name(p.name + '.tmp')
    t.write_text(json.dumps(value, indent=2, ensure_ascii=False, allow_nan=False) + '\n')
    os.replace(t, p)

def config(path):
    c = json.loads(Path(path).read_text())
    c['_config_path'] = str(Path(path).resolve())
```

```python
def validate_splits(*datasets):

                'guard_seconds', 'source_sha256', 'frame_index', 'metadata', 'handcrafted')

        if not np.array_equal(d['y'], np.array([grade(v) for v in d['ssim']])):
            raise ValueError('SSIM/label boundary mismatch')
        for k in ('metadata', 'handcrafted', 'segment_start', 'segment_duration', 'guard_seconds'):
            if not np.isfinite(d[k]).all():
                raise ValueError(k + ' not finite')
        if np.any(d['segment_duration'] <= 0) or np.any(d['guard_seconds'] < 0):
            raise ValueError('invalid temporal span')
        for k in ('video_id', 'group_id', 'sample_id', 'source_sha256'):
            if '' in set(d[k].astype(str)):
                raise ValueError('empty ' + k)
        ss = set(d['sample_id'].astype(str))
        if len(ss) != n:
            raise ValueError('duplicate sample ID')
        groups.append(set(d['group_id'].astype(str))); samples.append(ss)
        spans.append(set(zip(d['video_id'].astype(str), d['source_sha256'].astype(str),
                             d['segment_start'].astype(float), d['segment_duration'].astype(float), d['guard_seconds'].astype(float))))
    for i in range(len(datasets)):
        for j in range(i):
            if groups[i] & groups[j] or samples[i] & samples[j]:
                raise ValueError('segment or sample crosses splits')
            for vi, hi, si, di, gi in spans[i]:
                for vj, hj, sj, dj, gj in spans[j]:

                        continue

                    if distance < max(gi, gj) - 1e-6:
```

```python
import json
import os
import time
import urllib.error
import urllib.request
from pathlib import Path
from ..common import sha256, write_json
from .mpd import parse_mpd

def head(url):
    with urllib.request.urlopen(urllib.request.Request(url, method='HEAD'), timeout=45) as r:
        return dict(url=r.url, size_bytes=int(r.headers['Content-Length']) if r.headers.get('Content-Length') else None,
                    etag=r.headers.get('ETag'), last_modified=r.headers.get('Last-Modified'))

def download(url, destination):
    dst = Path(destination); dst.parent.mkdir(parents=True, exist_ok=True)
    remote = head(url); receipt = dst.with_suffix(dst.suffix + '.receipt.json')
    if dst.exists():
        if not receipt.exists():
            raise ValueError(f'Existing file has no download receipt: {dst}; register local media separately or move it aside')
        saved = json.loads(receipt.read_text())
        if saved['url'] != url or dst.stat().st_size != saved['size_bytes'] or sha256(dst) != saved['sha256']:
            raise ValueError(f'Existing download identity/hash mismatch: {dst}')
        print('verified existing', dst, flush=True); return saved
    part = dst.with_suffix(dst.suffix + '.part'); meta = part.with_suffix(part.suffix + '.json')
    if part.exists():
        if not meta.exists() or json.loads(meta.read_text()) != remote:
            raise ValueError(f'Partial download source changed: {part}; move it aside before retrying')
    else:
        write_json(meta, remote)
    for attempt in range(3):
        start = part.stat().st_size if part.exists() else 0
        total = remote['size_bytes']
        if total is not None and start > total:
            raise ValueError('partial download exceeds remote length')
        if total is not None and start == total:
            break
        headers = {'User-Agent': 'DASH-AVC4-research/1.0', 'Accept-Encoding': 'identity'}
        if start:
            headers['Range'] = f'bytes={start}-'
            if remote['etag']:
                headers['If-Range'] = remote['etag']
        try:
            with urllib.request.urlopen(urllib.request.Request(url, headers=headers), timeout=60) as response:
                if start and (response.status != 206 or not response.headers.get('Content-Range', '').startswith(f'bytes {start}-')):
                    raise ValueError('server did not honor resumable range; partial file kept')
                with part.open('ab' if start else 'wb') as f:
                    printed = start
                    while True:
                        chunk = response.read(1024 * 1024)
                        if not chunk:
                            break
                        f.write(chunk); start += len(chunk)
                        if start - printed >= 128 * 1024 * 1024:
                            print(dst.name, f'{start / 1e9:.2f} GB', flush=True); printed = start
            if total is not None and start != total:
                raise OSError(f'incomplete transfer: {start}/{total}')
            break
        except (OSError, urllib.error.URLError):
            if attempt == 2:
                raise
            time.sleep(1)
    os.replace(part, dst); meta.unlink(missing_ok=True)
    saved = dict(url=url, size_bytes=dst.stat().st_size, sha256=sha256(dst), remote=remote)
    write_json(receipt, saved); print('downloaded', dst, flush=True)
    return saved

def acquire(cfg, work, references=False, plan=False):
    work = Path(work); videos = []
    for v in cfg['videos']:
        root = work / 'media' / v['id']; mpd = root / 'manifest.mpd'
        download(v['mpd_url'], mpd)
        reps = parse_mpd(mpd, v['mpd_url'])
        selected = cfg.get('representation_indices', 'all')
        if selected != 'all':
            if not selected or len(set(selected)) != len(selected) or any(i < 0 or i >= len(reps) for i in selected):
                raise ValueError('invalid representation selection')
            reps = [r for r in reps if r.index in selected]
        info = dict(id=v['id'], mpd=str(mpd.resolve()), mpd_url=v['mpd_url'], reference_url=v['reference_url'],
                    reference_path=str((root / 'reference.y4m.xz').resolve()), representations=[])
        for r in reps:
            path = root / f'rep_{r.index:02d}.mp4'
            record = dict(r.__dict__, path=str(path.resolve()))
            record['remote'] = head(r.url) if plan else download(r.url, path)
            info['representations'].append(record)
        if references or plan:
            info['reference_remote'] = head(v['reference_url']) if plan else download(v['reference_url'], info['reference_path'])
        videos.append(info)
    result = dict(status='DOWNLOAD_PLAN' if plan else 'DOWNLOADED', videos=videos)
    if plan:
        result['known_download_bytes'] = sum((v.get('reference_remote', {}).get('size_bytes') or 0) +
                                            sum(r['remote'].get('size_bytes') or 0 for r in v['representations']) for v in videos)
        print('Known total download GB:', result['known_download_bytes'] / 1e9)
```

```python
"""Local SSIM teacher: 11x11 Gaussian, sigma1.5, valid, population moments."""
import numpy as np
from numpy.lib.stride_tricks import sliding_window_view
from ..spec import SSIM_THRESHOLDS

def grade(score):
    s = float(score)
    if not np.isfinite(s) or not -1 <= s <= 1:
        raise ValueError('finite SSIM in [-1,1] required')
    poor, great, excellent = SSIM_THRESHOLDS
    return 0 if s >= excellent else 1 if s >= great else 2 if s > poor else 3

def ssim_local(reference, distorted):
    a = np.asarray(reference, np.float64); b = np.asarray(distorted, np.float64)
    if a.shape != (28, 28) or b.shape != a.shape:
        raise ValueError('SSIM needs aligned 28x28 patches')
    if not np.isfinite(a).all() or not np.isfinite(b).all() or min(a.min(), b.min()) < 0 or max(a.max(), b.max()) > 255:
        raise ValueError('Y code range 0..255')
    g = np.exp(-np.arange(-5, 6, dtype=float) ** 2 / (2 * 1.5 ** 2)); g /= g.sum()
    window = g[:, None] * g[None, :]
    blur = lambda x: np.einsum('ijab,ab->ij', sliding_window_view(x, (11, 11)), window)
    ma, mb = blur(a), blur(b)
    va = np.maximum(0, blur(a * a) - ma * ma); vb = np.maximum(0, blur(b * b) - mb * mb)
    cov = blur(a * b) - ma * mb
    c1, c2 = (.01 * 255) ** 2, (.03 * 255) ** 2
    return float(np.clip((((2 * ma * mb + c1) * (2 * cov + c2)) /
                         ((ma * ma + mb * mb + c1) * (va + vb + c2))).mean(), -1, 1))
```

```python
def _read_exact(f, n):
    parts = []; remaining = n
    while remaining:
        a = f.read(remaining)
        if not a:
            raise EOFError('truncated Y4M frame payload')
        parts.append(a); remaining -= len(a)
    return b''.join(parts)

def _skip(f, n):
    while n:
        size = min(n, 1024 * 1024); _read_exact(f, size); n -= size

                Image.fromarray(y).save(out / f'{current:08d}.png')
```

```python
"""Parse this dataset's AVC onDemand SegmentBase MPD and actual ISO-BMFF SIDX."""
from dataclasses import dataclass, asdict
from fractions import Fraction
import struct
import xml.etree.ElementTree as ET
from urllib.parse import urljoin

@dataclass(frozen=True)
class Representation:
    index: int
    id: str
    width: int
    height: int
    bandwidth: int
    fps: float
    url: str
    index_start: int
    index_end: int

def parse_mpd(path, url):
    root = ET.parse(path).getroot(); ns = {'m': root.tag.split('}')[0][1:]}
    if root.get('type') != 'static' or 'on-demand' not in root.get('profiles', ''):
        raise ValueError('static ISO onDemand MPD required')
    periods = root.findall('m:Period', ns)
    if len(periods) != 1:
        raise ValueError('one Period required')
    rows = []
    for a in periods[0].findall('m:AdaptationSet', ns):
        for r in a.findall('m:Representation', ns):
            codec = r.get('codecs', a.get('codecs', ''))
            if not codec.startswith(('avc1', 'avc3')):
                continue
            base, seg = r.find('m:BaseURL', ns), r.find('m:SegmentBase', ns)
            if base is None or seg is None or not seg.get('indexRange'):
                raise ValueError('AVC Representation requires BaseURL and SegmentBase indexRange')
            b0 = root.find('m:BaseURL', ns); b1 = periods[0].find('m:BaseURL', ns); b2 = a.find('m:BaseURL', ns)
            resolved = url
            for b in (b0, b1, b2, base):
                if b is not None:
                    resolved = urljoin(resolved, b.text.strip())
            begin, end = map(int, seg.get('indexRange').split('-'))
            rows.append(dict(id=r.get('id'), width=int(r.get('width')), height=int(r.get('height')),
                             bandwidth=int(r.get('bandwidth')), fps=float(Fraction(r.get('frameRate', a.get('frameRate', '24')))),
                             url=resolved, index_start=begin, index_end=end))
    rows.sort(key=lambda r: r['bandwidth'])
    if not rows:
        raise ValueError('no AVC representations')
    return [Representation(index=i, **r) for i, r in enumerate(rows)]

def parse_sidx(data, absolute_start=0):
    """Resolve direct references. Offsets are absolute bytes; duration is never bitrate-derived."""
    pos = 0
    while pos + 8 <= len(data):
        size, kind = struct.unpack_from('>I4s', data, pos); header = 8
        if size == 1:
            size = struct.unpack_from('>Q', data, pos + 8)[0]; header = 16
        if size == 0:
            size = len(data) - pos
        if size < header or pos + size > len(data):
            raise ValueError('truncated ISO box')
        if kind != b'sidx':
            pos += size; continue
        q = pos + header; version = data[q]; q += 4
        if version not in (0, 1):
            raise ValueError('unknown SIDX version')
        _, scale = struct.unpack_from('>II', data, q); q += 8
        if not scale:
            raise ValueError('zero SIDX timescale')
        fmt = '>II' if version == 0 else '>QQ'
        earliest, first = struct.unpack_from(fmt, data, q); q += struct.calcsize(fmt)
        _, count = struct.unpack_from('>HH', data, q); q += 4
        offset = absolute_start + pos + size + first; ticks = earliest; result = []
        if q + count * 12 > pos + size:
            raise ValueError('truncated SIDX entries')
        for i in range(count):
            length, duration, sap = struct.unpack_from('>III', data, q); q += 12
            if length >> 31:
                raise ValueError('hierarchical SIDX unsupported; do not invent segment durations')
            length &= 0x7fffffff
            if not length or not duration:
                raise ValueError('empty segment')
            result.append(dict(segment_index=i, start=ticks / scale, duration=duration / scale,
                               offset=offset, size_bytes=length, starts_with_sap=bool(sap >> 31)))
            ticks += duration; offset += length
        return result
    raise ValueError('SIDX absent at MPD indexRange')

def local_segments(path, rep):
    with open(path, 'rb') as f:
        f.seek(rep.index_start); data = f.read(rep.index_end - rep.index_start + 1)
    rows = parse_sidx(data, rep.index_start)
    if any(r['offset'] + r['size_bytes'] > __import__('os').path.getsize(path) for r in rows):
        raise ValueError('SIDX references exceed media file')
    if any(abs(r['duration'] - 2) > 1 / rep.fps + 1e-6 for r in rows[:-1]):
        raise ValueError('this configuration requires 2-second segments; final short segment allowed')
    return rows
```

```python
import numpy as np

def coordinates(height, width, grid):
    if min(height, width) < 28:
        raise ValueError('frame smaller than patch')
    result = [(int(round((height - 28) * yy)), int(round((width - 28) * xx))) for yy, xx in grid]

def controls(x, bitrate, width, height, fps):
    a = x.astype(float) / 255
    meta = np.array([np.log2(max(bitrate / (width * height * fps), 1e-9)), np.log2(width), np.log2(height)])
    spatial = np.array([a.mean(), a.std(), np.abs(np.diff(a, axis=0)).mean(), np.abs(np.diff(a, axis=1)).mean(),
                        (np.diff(a, 2, axis=0) ** 2).mean(), (np.diff(a, 2, axis=1) ** 2).mean()])
    return meta, np.r_[meta, spatial]
```

```python
"""Blocked temporal prototype split. All representations of a scene interval stay together."""
import numpy as np

def assignments(video_id, duration, cfg):
    block = float(cfg['block_seconds']); guard = float(cfg['guard_seconds'])
    if block <= 2 * guard or guard < 0:
        raise ValueError('block_seconds must exceed two guard margins')
    n = int(np.ceil(duration / block))
    if n < 7:
        raise ValueError('at least seven temporal blocks per movie required')
    # Stable movie-specific seed, never Python hash().
    import hashlib
    seed = int(cfg['seed']) + int(hashlib.sha256(video_id.encode()).hexdigest()[:8], 16)
    order = np.random.default_rng(seed).permutation(n)
    nv = max(1, round(n * float(cfg['validation_fraction'])))
    nt = max(1, round(n * float(cfg['test_fraction'])))
    if nv + nt >= n:
        raise ValueError('no training blocks')
    table = {int(i): 'train' for i in order}
    table.update({int(i): 'validation' for i in order[:nv]})
    table.update({int(i): 'test' for i in order[nv:nv + nt]})
    return table

def segment_split(segment, duration, table, cfg):
    block = float(cfg['block_seconds']); guard = float(cfg['guard_seconds'])
    start, end = segment['start'], segment['start'] + segment['duration']
    k = int(start // block)
    if k not in table or start < k * block + guard or end > min((k + 1) * block - guard, duration):
        return None
    return table[k]
```

## src/dash_cnn/evaluation/metrics.py

```python
"""Four-class metrics with explicit support, adjacent errors and severe->excellent misses."""
import numpy as np
from ..spec import NUM_CLASSES, CLASSES

def metrics(y, pred, video_id=None):
    y, p = np.asarray(y), np.asarray(pred)
    if y.shape != p.shape or y.ndim != 1 or not len(y):
        raise ValueError('nonempty matched 1D labels')
    if min(y.min(), p.min()) < 0 or max(y.max(), p.max()) >= NUM_CLASSES:
        raise ValueError('class range')
    cm = np.zeros((NUM_CLASSES, NUM_CLASSES), np.int64); np.add.at(cm, (y, p), 1)
    support = cm.sum(1); den = support + cm.sum(0)
    f1 = np.divide(2 * np.diag(cm), den, out=np.zeros(NUM_CLASSES), where=den > 0)
    recall = np.divide(np.diag(cm), support, out=np.zeros(NUM_CLASSES), where=support > 0)
    r = dict(n=len(y), accuracy=float(np.trace(cm) / len(y)), macro_f1=float(f1.mean()),
             class_names=list(CLASSES), class_support=support.tolist(), class_f1=f1.tolist(),
             class_recall=recall.tolist(), confusion=cm.tolist(),
             severe_missed_as_excellent=None if not support[3] else float(cm[3, 0] / support[3]),
             mean_absolute_grade_error=float(np.abs(y - p).mean()),
             all_classes_present=bool(np.all(support > 0)))
    if video_id is not None:
        ids = np.asarray(video_id).astype(str)
        r['videos'] = {k: metrics(y[ids == k], p[ids == k]) for k in sorted(set(ids))}
    return r
```

```python
import numpy as np
from .arithmetic import A,B,G,BLOCKS,conv_direct,assert_i32
from ..spec import VARIANTS,code_layout,NUM_CLASSES

def effective(q,variant='final'):
    return np.asarray(q,np.int64)*(2**code_layout(VARIANTS[variant])[2])

def rne_shift(x,shift):
    x=np.asarray(x,np.int64);s=int(shift)
    if not 0<=s<=31:raise ValueError('shift outside 0..31')
    if not s:return x.copy()
    base=x>>s;rem=x-(base<<s);half=1<<(s-1)
    return base+((rem>half)|((rem==half)&((base&1)!=0)))

def dwm_tile(x,q,variant='final'):
    y=np.zeros((2,2),np.int64);delta=code_layout(VARIANTS[variant])[2]
    for r,c,kr,kc in BLOCKS:
        if VARIANTS[variant]['mask'] and r==3 and c==3:continue
        d=B[kr]@x[r:r+kr+1,c:c+kc+1]@B[kc].T
        u=G[kr]@q[r:r+kr,c:c+kc]@G[kc].T
        z=A[kr]@(d*u)@A[kc].T
        div=(2 if kr==3 else 1)*(2 if kc==3 else 1)
        if np.any(z%div):raise AssertionError('non-exact inverse transform')
        y+=(z//div)*(1<<int(delta[r,c]))
    return y

def conv_dwm(x,q,variant='final'):
    oc,ic=q.shape[:2];h,w=x.shape[-2]-4,x.shape[-1]-4
    if h%2 or w%2:raise ValueError('2x2 tiles require even output dimensions')
    out=np.zeros((oc,h,w),np.int64)
    for o in range(oc):
        for r in range(0,h,2):
            for c in range(0,w,2):
                for i in range(ic):out[o,r:r+2,c:c+2]+=dwm_tile(x[i,r:r+6,c:c+6],q[o,i],variant)
    return out

def validate(p,variant='final'):
    shapes={'q1':(6,1,5,5),'q2':(12,6,5,5),'qfc':(NUM_CLASSES,192),
            'b1':(6,),'b2':(12,),'bfc':(NUM_CLASSES,)}
    for k,sh in shapes.items():
        a=np.asarray(p[k])
        if a.shape!=sh or not np.issubdtype(a.dtype,np.integer):raise ValueError(k+' shape/type')
    v=VARIANTS[variant];lo,hi,_=code_layout(v)
    for k in ['q1','q2']:
        if np.any(p[k]<lo) or np.any(p[k]>hi):raise ValueError(k+' invalid mask/weight code')
    if p['qfc'].min()<-(1<<(v['fc_bits']-1)) or p['qfc'].max()>=(1<<(v['fc_bits']-1)):raise ValueError('FC code')
    for k in ['s1','s2']:
        a=np.asarray(p[k])
        if a.ndim or not np.issubdtype(a.dtype,np.integer) or not 0<=int(a)<=31:raise ValueError(k+' shift')
    layers=[(effective(p['q1'],variant),p['b1'],255)]
    layers += [(effective(p['q2'],variant),p['b2'],(1<<v['a1_bits'])-1)]
    layers += [(p['qfc'],p['bfc'],(1<<v['a2_bits'])-1)]
    for w,b,mx in layers:
        w=w.reshape(len(b),-1);assert_i32(b)
        assert_i32(mx*np.minimum(w,0).sum(1)+b);assert_i32(mx*np.maximum(w,0).sum(1)+b)

def pool(raw,b,shift,bits):
    raw=raw+b[:,None,None];assert_i32(raw)
    q=np.clip(rne_shift(raw,shift),0,(1<<bits)-1)
    ch,h,w=q.shape
    return q.reshape(ch,h//2,2,w//2,2).max((2,4))

def infer(x,p,profile=0,method='direct',variant='final'):
    validate(p,variant);x=np.asarray(x)
    if x.shape!=(28,28) or not np.issubdtype(x.dtype,np.integer) or x.min()<0 or x.max()>255:raise ValueError('uint8-range 28x28 required')
    if profile not in (0,1):raise ValueError('AVC shared model accepts profile 0 or 1; unsupported codecs need host fallback')
    v=VARIANTS[variant]
    if method=='direct':fn=lambda x,q:conv_direct(x,effective(q,variant))
    elif method=='dwm':fn=lambda x,q:conv_dwm(x,q,variant)
    else:raise ValueError('method')
    p1=pool(fn(x[None],p['q1']),p['b1'],p['s1'],v['a1_bits'])
    p2=pool(fn(p1,p['q2']),p['b2'],p['s2'],v['a2_bits'])
    flat=p2.ravel();scores=p['bfc'].copy()
    for i in range(0,192,40):scores+=p['qfc'][:,i:i+40]@flat[i:i+40]
    assert_i32(scores)
    return dict(pool1=p1,pool2=p2,scores=scores,decision=int(np.argmax(scores)))

def load(path):
    with np.load(path,allow_pickle=False) as f:return {k:f[k] for k in f.files}

def export(p,directory,variant='final'):
    validate(p,variant);d=Path(directory);d.mkdir(parents=True,exist_ok=True)
    np.savez_compressed(d/'weights_int.npz',**p)
    for k in ['q1','q2','qfc','b1','b2','bfc']:
        bits=32 if k.startswith('b') else 8
        (d/(k+'.mem')).write_text(''.join(f'{int(x)&((1<<bits)-1):0{bits//4}x}\n' for x in p[k].ravel()))
    (d/'params.svh').write_text(f'`define DASH_S1 {int(p["s1"])}\n`define DASH_S2 {int(p["s2"])}\n')
    (d/'manifest.json').write_text(json.dumps(dict(variant=variant,shape='HWC input; OIHW kernel; CHW flatten',
        s1=int(p['s1']),s2=int(p['s2']),rounding='nearest_even',weight_words='byte sign-extended; logical packing is separate',
        profiles=['shared_avc'],classes=NUM_CLASSES,codec_banks=1,provenance='UNSPECIFIED_NOT_A_TRAINED_CLAIM'),indent=2)+'\n')
```

```python
"""Four-class AVC CNN. Shared weights; no codec adapter in the first experiment."""
import math
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
from ..spec import VARIANTS,code_layout,NUM_CLASSES

def fake(v,lo,hi):
    z=torch.clamp(v,min=lo,max=hi)
    return z+(torch.round(z)-z).detach()

class DASHCNN(nn.Module):
    def __init__(self,variant='final'):
        super().__init__();self.variant=variant;self.v=VARIANTS[variant]
        coords=[(r,c) for r in range(5) for c in range(5) if not self.v['mask'] or r<3 or c<3]
        self.n=len(coords);sc=torch.zeros(self.n,5,5)
        for i,(r,c) in enumerate(coords):sc[i,r,c]=1
        self.register_buffer('scatter',sc)
        lo,hi,delta=code_layout(self.v)
        for k,a in [('lo',lo),('hi',hi),('delta',delta)]:self.register_buffer(k,torch.tensor(a,dtype=torch.float32))
        self.w1=nn.Parameter(torch.empty(6,1,self.n));self.b1=nn.Parameter(torch.zeros(6))
        self.w2=nn.Parameter(torch.empty(12,6,self.n));self.b2=nn.Parameter(torch.zeros(12))
        self.wfc=nn.Parameter(torch.empty(NUM_CLASSES,192));self.bfc=nn.Parameter(torch.zeros(NUM_CLASSES))
        for w in [self.w1,self.w2,self.wfc]:nn.init.kaiming_uniform_(w,a=math.sqrt(5))
        self.quant=None
    def kernels(self,profile):
        if profile not in (0,1):raise ValueError('AVC shared only')
        w=self.w2;b=self.b2
        return torch.einsum('oik,krc->oirc',self.w1,self.scatter),torch.einsum('oik,krc->oirc',w,self.scatter),b
    def qw(self,w,e):
        s=2.**e*2.**self.delta
        return fake(w/s,self.lo,self.hi)*s
    def qb(self,b,e):return fake(b/2.**e,-2**31,2**31-1)*2.**e
    def qa(self,x,e,bits):return fake(x/2.**e,0,2**bits-1)*2.**e
    def forward(self,x,profile=0,maps=False):
        if x.ndim!=4 or tuple(x.shape[1:])!=(1,28,28):raise ValueError('N,1,28,28 required')
        w1,w2,b2=self.kernels(profile);b1=self.b1;wf=self.wfc;bf=self.bfc
        if self.quant is not None:
            q=self.quant;w1=self.qw(w1,q['ew1']);w2=self.qw(w2,q['ew2'])
            lim=2**(self.v['fc_bits']-1);wf=fake(wf/2.**q['ewfc'],-lim,lim-1)*2.**q['ewfc']
            b1=self.qb(b1,-8+q['ew1']);b2=self.qb(b2,q['ea1']+q['ew2']);bf=self.qb(bf,q['ea2']+q['ewfc'])
        y=F.conv2d(x,w1,b1)
        y=F.relu(y) if self.quant is None else self.qa(y,q['ea1'],self.v['a1_bits'])
        p1=F.max_pool2d(y,2);y=F.conv2d(p1,w2,b2)
        y=F.relu(y) if self.quant is None else self.qa(y,q['ea2'],self.v['a2_bits'])
        p2=F.max_pool2d(y,2);scores=F.linear(p2.flatten(1),wf,bf)
        return (scores,p1,p2) if maps else scores
    def train_base(self):
        for p in self.parameters():p.requires_grad_(True)
    @torch.no_grad()
    def calibrate(self,x):
        exp=lambda x:math.ceil(math.log2(max(float(x),1e-12)))
        w1,w2,_=self.kernels(0);cap=self.hi.clamp(min=1)*2.**self.delta
        e1=exp((w1.abs()/cap).max());e2=exp((w2.abs()/cap).max())
        ef=exp(self.wfc.abs().max()/(2**(self.v['fc_bits']-1)-1))
        y=F.conv2d(x,self.qw(w1,e1),self.qb(self.b1,-8+e1))
        a1=max(exp(y.clamp(min=0).max()/(2**self.v['a1_bits']-1)),-8+e1)
        p1=F.max_pool2d(self.qa(y,a1,self.v['a1_bits']),2)
        y=F.conv2d(p1,self.qw(w2,e2),self.qb(self.b2,a1+e2))
        a2=max(exp(y.clamp(min=0).max()/(2**self.v['a2_bits']-1)),a1+e2)
        self.quant=dict(ew1=e1,ew2=e2,ewfc=ef,ea1=a1,ea2=a2)
        if not 0<=a1+8-e1<=31 or not 0<=a2-a1-e2<=31:raise ValueError('unsupported shift')
        return self.quant
    @torch.no_grad()
    def export_integer(self):
        if self.quant is None:raise ValueError('quantization not calibrated')
        q=self.quant
        arr=lambda a:a.detach().cpu().double().numpy()
        lo,hi,delta=code_layout(self.v)
        enc=lambda w,e:np.clip(np.rint(arr(w)/(2.**e*2.**delta)),lo,hi).astype(np.int64)
        def bias(b,e):
            z=np.rint(arr(b)/2.**e)
            if z.min()<-2**31 or z.max()>2**31-1:raise OverflowError('bias')
            return z.astype(np.int64)
        ks=[self.kernels(0)];lim=2**(self.v['fc_bits']-1)
        p=dict(q1=enc(ks[0][0],q['ew1']),q2=enc(ks[0][1],q['ew2']),
            qfc=np.clip(np.rint(arr(self.wfc)/2.**q['ewfc']),-lim,lim-1).astype(np.int64),
            b1=bias(self.b1,-8+q['ew1']),b2=bias(ks[0][2],q['ea1']+q['ew2']),
            bfc=bias(self.bfc,q['ea2']+q['ewfc']),s1=np.array(q['ea1']+8-q['ew1']),s2=np.array(q['ea2']-q['ea1']-q['ew2']))
        from ..integer.model import validate
        validate(p,self.variant);return p
```

```python
"""FP32 -> train-only calibration -> QAT -> deployment-integer validation -> export."""
import copy
import csv
import json
from pathlib import Path
import random
import numpy as np
import torch
from torch.nn import functional as F
from ..spec import NUM_CLASSES
from ..models.cnn import DASHCNN
from ..data.audit import load, validate_splits
from ..integer.model import infer, export
from ..evaluation.metrics import metrics
from ..common import write_json, sha256

@torch.no_grad()
def evaluate_torch(model, d):
    model.eval(); pred = []
    device, dtype = next(model.parameters()).device, next(model.parameters()).dtype
    for i in range(0, len(d['x']), 128):
        x = torch.from_numpy(d['x'][i:i + 128])[:, None].to(device, dtype) / 256
        pred.extend(model(x).argmax(1).cpu().numpy())
    return metrics(d['y'], np.asarray(pred), d['video_id'])

def evaluate_integer(model, d):
    p = model.export_integer()
    pred = np.array([infer(x, p, variant=model.variant)['decision'] for x in d['x']])
    return metrics(d['y'], pred, d['video_id'])

def fit(model, td, vd, *, epochs, lr, batch_size, seed, quantized):
    if epochs < 1 or batch_size < 1 or lr <= 0:
        raise ValueError('positive epochs/batch/lr')
    opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
    rng = np.random.default_rng(seed)
    counts = np.bincount(td['y'], minlength=NUM_CLASSES)
    if np.any(counts == 0):
        raise ValueError('train needs all four SSIM classes; inspect dataset_audit.json, keep thresholds fixed')
    probs = 1. / counts[td['y']]; probs /= probs.sum()
    device = next(model.parameters()).device; best = -1.; state = None; history = []
    for epoch in range(epochs):
        ix = rng.choice(len(td['x']), len(td['x']), replace=True, p=probs)
        model.train(); total_loss = 0.; total = 0
        for i in range(0, len(ix), batch_size):
            ids = ix[i:i + batch_size]
            x = torch.from_numpy(td['x'][ids])[:, None].to(device).float() / 256
            y = torch.from_numpy(td['y'][ids]).to(device).long()
            opt.zero_grad(); loss = F.cross_entropy(model(x), y); loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), 5.)
            opt.step(); total_loss += float(loss.detach()) * len(ids); total += len(ids)
        m = evaluate_integer(model, vd) if quantized else evaluate_torch(model, vd)
        score = m['macro_f1']
        row = dict(stage='qat' if quantized else 'fp32', epoch=epoch + 1, loss=total_loss / total,
                   validation_accuracy=m['accuracy'], validation_macro_f1=score,
                   validation_metrics=m, selection_arithmetic='integer' if quantized else 'float32')
        history.append(row); print(row['stage'], epoch + 1, 'loss', round(row['loss'], 5), 'val_F1', round(score, 5), flush=True)
        if score > best:
            best = score; state = copy.deepcopy(model.state_dict())
    model.load_state_dict(state); return history

@torch.no_grad()
def audit_export(model, data, params):
    m = copy.deepcopy(model).cpu().double().eval(); q = m.quant; n = 0
    for i in np.linspace(0, len(data['x']) - 1, min(6, len(data['x'])), dtype=int):
        x = data['x'][i]; ref = infer(x, params, variant=m.variant); win = infer(x, params, method='dwm', variant=m.variant)
        scores, p1, p2 = m(torch.tensor(x, dtype=torch.float64)[None, None] / 256, maps=True)
        for key, tensor, exp in [('pool1', p1, q['ea1']), ('pool2', p2, q['ea2']), ('scores', scores, q['ea2'] + q['ewfc'])]:
            got = np.rint(tensor[0].numpy() / 2. ** exp).astype(np.int64)
            if not np.array_equal(ref[key], win[key]) or not np.array_equal(got, ref[key]):
                raise AssertionError('QAT/NumPy/direct/PFDW export mismatch: ' + key)
        n += 1
    return n

def train(cfg, work, device='cpu', base_epochs=None, qat_epochs=None, variant=None):
    work = Path(work); c = dict(cfg['training'])
    if base_epochs is not None: c['base_epochs'] = base_epochs
    if qat_epochs is not None: c['qat_epochs'] = qat_epochs
    if variant is not None: c['variant'] = variant
    td, vd = [load(work / 'data' / (s + '.npz')) for s in ('train', 'validation')]
    audit = validate_splits(td, vd)
    if not all(np.bincount(td['y'], minlength=NUM_CLASSES)):
        raise ValueError('train needs all four classes; inspect dataset_audit.json before training')
    if not all(np.bincount(vd['y'], minlength=NUM_CLASSES)):
        raise ValueError('validation needs all four classes for model selection')
    seed = int(c['seed']); random.seed(seed); np.random.seed(seed); torch.manual_seed(seed)
    torch.set_num_threads(2)
    if device.startswith('cuda') and not torch.cuda.is_available():
        raise ValueError('CUDA requested but unavailable; use --device cpu')
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed); torch.backends.cudnn.benchmark = False; torch.backends.cudnn.deterministic = True
    out = work / 'models' / c['variant']
    if out.exists():
        raise ValueError('model output already exists; use a new work directory or a different variant')
    out.mkdir(parents=True)
    model = DASHCNN(c['variant']).to(device); model.train_base()
    hist = fit(model, td, vd, epochs=c['base_epochs'], lr=c['base_lr'], batch_size=c['batch_size'], seed=seed, quantized=False)
    torch.save(dict(state=model.cpu().state_dict(), variant=model.variant), out / 'fp32.pt'); model.to(device)
    idx = np.random.default_rng(seed).choice(len(td['x']), min(c['calibration_samples'], len(td['x'])), replace=False)
    # Bound calibration memory on GPU; no validation/test samples are used.
    model.calibrate(torch.from_numpy(td['x'][idx])[:, None].to(device).float() / 256)
    hist += fit(model, td, vd, epochs=c['qat_epochs'], lr=c['qat_lr'], batch_size=c['batch_size'], seed=seed + 1, quantized=True)
    params = model.export_integer(); crosschecks = audit_export(model, vd, params)
    export(params, out / 'integer', model.variant)
    synthetic = bool(np.asarray(td.get('synthetic_smoke', False)).item()) or bool(np.asarray(vd.get('synthetic_smoke', False)).item())
    status = 'SYNTHETIC_SMOKE_ONLY' if synthetic else 'TRAINED_CANDIDATE_TEST_AND_PPA_PENDING'
    manifest = json.loads((out / 'integer' / 'manifest.json').read_text())
    manifest.update(provenance=status, seed=seed, train_sha256=sha256(work / 'data/train.npz'),
                    validation_sha256=sha256(work / 'data/validation.npz'), quant=model.quant)
    write_json(out / 'integer' / 'manifest.json', manifest)
    torch.save(dict(state=model.cpu().state_dict(), quant=model.quant, variant=model.variant, classes=NUM_CLASSES), out / 'qat.pt')
    with (out / 'history.csv').open('w', newline='') as f:
        keys = ('stage', 'epoch', 'loss', 'validation_accuracy', 'validation_macro_f1')
        w = csv.DictWriter(f, fieldnames=keys, extrasaction='ignore'); w.writeheader(); w.writerows(hist)
    report = dict(status=status, training_config=c, split_audit=audit, history=hist, integer_crosschecks=crosschecks,
                  final_validation=evaluate_integer(model, vd), quant=model.quant, torch_version=torch.__version__,
                  test='NOT_READ', ppa='NOT_MEASURED', codec_adaptation='NOT_USED_AVC_SHARED')
    write_json(out / 'training_report.json', report); print(status); return out
```
